b) Can the `getAccessLevel()` function be simplified to reduce its cyclomatic complexity, while preserving its existing behaviour?

```js
function getAccessLevel(role, isVerified, department, hasSecurityClearance) {
  let access = "none";

  if (role === "admin") {
    if (hasSecurityClearance) {
      access = "full";
    } else {
      access = "limited";
    }
  } else if (role === "manager") {
    if (department === "IT") {
      access = "high";
    } else {
      access = "medium";
    }
  } else {
    if (isVerified) {
      access = "basic";
    } else {
      access = "restricted";
    }
  }

  return access;
}
```
